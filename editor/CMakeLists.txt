add_library(EditorTools STATIC src/editor.cpp src/scene_document.cpp src/grid_tools.cpp)
target_include_directories(EditorTools PUBLIC include PRIVATE src)
target_link_libraries(EditorTools PUBLIC Engine ImGuiSupport PRIVATE SDL3::SDL3 Vulkan::Vulkan)
target_compile_features(EditorTools PUBLIC cxx_std_20)
set_target_properties(EditorTools PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(EditorTools PRIVATE /W4 /permissive-)
else()
    target_compile_options(EditorTools PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(GameDebug STATIC src/debug/navigation_debug.cpp)
target_include_directories(GameDebug PUBLIC src)
target_link_libraries(GameDebug PUBLIC GameLogic EditorTools PRIVATE DearImGui)
set_target_properties(GameDebug PROPERTIES CXX_EXTENSIONS OFF)

add_executable(Editor src/main.cpp)
target_link_libraries(Editor PRIVATE GameDebug EditorTools)
set_target_properties(Editor PROPERTIES OUTPUT_NAME GrimlockEditor
    RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>" CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(GameDebug PRIVATE /W4 /permissive-)
    target_compile_options(Editor PRIVATE /W4 /permissive-)
else()
    target_compile_options(GameDebug PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(Editor PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    add_custom_command(TARGET Editor POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:Editor> VERBATIM)
endif()
